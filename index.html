function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); // Электронды кестеге жаңа жол қосу sheet.appendRow([ data.date, // Уақыты мен датасы data.fullName, // Аты-жөні data.grade, // Сыныбы / Лауазымы data.score, // Упай (мысалы, 35/40) data.percent // Пайыз (%) ]); return ContentService.createTextOutput(JSON.stringify({"result": "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": error.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
